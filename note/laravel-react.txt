Laravel + React (File) + Create
React side
import axios from 'axios'
import React, { useEffect, useState } from 'react'

const Gallery = () => {
    const [gallery, setGallery] = useState([]);
    const url = import.meta.env.VITE_API_URL;
    const image_url = 'http://127.0.0.1:8000/storage/'
    const fetchImages = async () => {
        const response = await axios.get(url + '/api/images');
        console.log(response.data);
        
        if(response.status === 200){
            let data = response.data
            setGallery(data)
        }
    }

    useEffect(() => {
      fetchImages()
    }, [])
    
    
  return (
    <div>
        
        {
            gallery?.map((e, index) => (
                <img key={index} src={image_url + e.image}  />
            ))
        }
    </div>

    
  )
}

export default Gallery

Laravel + React (File) + Post
React side
import axios from 'axios'
import React, { useEffect, useState } from 'react'

const Gallery = () => {
   
    const [file, setFile] = useState(null)
    const url = import.meta.env.VITE_API_URL;
  
  
    
    const storeImage = async (e) => {
        e.preventDefault()
        const formData = new FormData();
        formData.append('image', file)
        const response = await axios.post(url + '/api/images/store', formData, {
            headers: {
                "Content-Type": "multipart/form-data",
            }
        })

        if(response.status === 200){
            const data = response.data
            setGallery(data)
            alert('image added ')
        }else{
            alert('something went wrong')
        }
    }
  return (
    <div>
        <form action="" onSubmit={storeImage} className='p-10 flex gap-x-3 items-center'>
            <input type="file" placeholder='insert an image' className='border px-2 py-1' onChange={(e) => setFile(e.target.files[0])}/>
            <button type='submit' className='px-4 py-1.5 bg-green-300 rounded-md '>Submit</button>
        </form>
    </div>

    
  )
}

export default Gallery


backend side
