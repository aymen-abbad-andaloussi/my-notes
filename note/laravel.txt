1 - go to

app/provider/appServiceProvider.php

2- add  your variable
3- share it with all views
4- make sure View is imported like this 
use Illuminate\Support\Facades\View;


    public function boot(): void
    {
        //
        $students = ['mohammed', 'amal', 'douaa', 'amhana'];
        $numberOfStudents = 4;
        View::share([
            'list' => $students,
            'class' => $numberOfStudents

    }
